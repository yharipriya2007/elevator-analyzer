% Acceleration due to gravity (m/s^2)
g = 9.81;

% Elevator and load masses (kg)
elevator_mass = 50;
loads = 10:10:100;

for load = loads
    total_mass = elevator_mass + load;
    tension = total_mass * g;

    fprintf('Load = %d kg\n', load);
    fprintf('Total Mass = %d kg\n', total_mass);
    fprintf('Pulley Tension = %.2f N\n', tension);
    fprintf('----------------------\n');
end
